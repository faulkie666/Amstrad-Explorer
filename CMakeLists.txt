if(CPCEXPLORER_ENABLE_FAST_ENGINE)
    if(MSVC)
        message(FATAL_ERROR "The reviewed cpu-z80 core uses GNU computed goto; configure the fast integration with Clang or GCC")
    endif()
    include(FetchContent)
    FetchContent_Declare(
        cpu_z80
        GIT_REPOSITORY https://github.com/michalhucik/z80-mz800.git
        GIT_TAG 687840df5b449e3b8ee31bf4cbc20f4ec6cf6d59
        GIT_SHALLOW TRUE
        PATCH_COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_LIST_DIR}/tools/patch_fast_z80_sram.py" <SOURCE_DIR>)
    FetchContent_MakeAvailable(cpu_z80)
    target_sources(cpc_core PRIVATE
        src/emulation/fast_z80_adapter.c
        "${cpu_z80_SOURCE_DIR}/cpu-z80/cpu/z80.c")
    set(CPCEXPLORER_FAST_SOURCE_DEFINITIONS "z80_reset=fastcore_z80_reset")
    if(CPCEXPLORER_FAST_RAM_ACCESS)
        list(APPEND CPCEXPLORER_FAST_SOURCE_DEFINITIONS CPU_Z80_RAM_FASTPATH=1)
        target_compile_definitions(cpc_core PRIVATE CPCEXPLORER_FAST_RAM_ACCESS=1)
    endif()
    if(CPCEXPLORER_FAST_RAM_VERIFY)
        if(NOT CPCEXPLORER_FAST_RAM_ACCESS OR CPCEXPLORER_BUILD_FIRMWARE)
            message(FATAL_ERROR "RAM fast-path verification requires a host build with RAM access enabled")
        endif()
        list(APPEND CPCEXPLORER_FAST_SOURCE_DEFINITIONS CPU_Z80_RAM_FASTPATH_VERIFY=1)
    endif()
    if(CPCEXPLORER_FAST_Z80_SRAM)
        list(APPEND CPCEXPLORER_FAST_SOURCE_DEFINITIONS CPU_Z80_SRAM_DISPATCH_TABLE=1)
    endif()
    if(CPCEXPLORER_STAGE4F6_SRAM_STEP)
        list(APPEND CPCEXPLORER_FAST_SOURCE_DEFINITIONS CPU_Z80_SRAM_EXECUTE_STEP=1)
    endif()
    set_source_files_properties(
        src/emulation/fast_z80_adapter.c
        "${cpu_z80_SOURCE_DIR}/cpu-z80/cpu/z80.c"
        PROPERTIES COMPILE_DEFINITIONS "${CPCEXPLORER_FAST_SOURCE_DEFINITIONS}")
    target_include_directories(cpc_core PRIVATE
        src/emulation "${cpu_z80_SOURCE_DIR}/cpu-z80")
    target_compile_definitions(cpc_core PRIVATE CPCEXPLORER_FAST_ENGINE=1)
    if(NOT CPCEXPLORER_FAST_PRESENTATION_DIVISOR STREQUAL "4" AND
       NOT CPCEXPLORER_FAST_PRESENTATION_DIVISOR STREQUAL "2")
        message(FATAL_ERROR "CPCEXPLORER_FAST_PRESENTATION_DIVISOR must be 4 or 2")
    endif()
    target_compile_definitions(cpc_core PRIVATE
        CPCEXPLORER_FAST_PRESENTATION_DIVISOR=${CPCEXPLORER_FAST_PRESENTATION_DIVISOR})
endif()

    if(CPCEXPLORER_STARTUP STREQUAL "KNIGHT_TYME" OR CPCEXPLORER_STARTUP STREQUAL "MENU" OR CPCEXPLORER_STARTUP STREQUAL "LIBRARY" OR CPCEXPLORER_STARTUP STREQUAL "LIBRARY_UPLOAD" OR CPCEXPLORER_STARTUP STREQUAL "LIBRARY_REMOVE" OR CPCEXPLORER_STARTUP STREQUAL "LIBRARY_CONTROLS" OR CPCEXPLORER_STARTUP STREQUAL "LIBRARY_DIAGNOSTIC" OR CPCEXPLORER_STARTUP STREQUAL "LIBRARY_DIAGNOSTIC_SRAM" OR CPCEXPLORER_STARTUP STREQUAL "LIBRARY_FAST")

           (CPCEXPLORER_FAST_RAM_ACCESS OR CPCEXPLORER_FAST_Z80_SRAM OR CPCEXPLORER_STAGE4F1_EXPERIMENT))
            message(FATAL_ERROR "Stage 4F.2 FAST_DISPLAY must use the unoptimised Stage 4E.2 Z80 path")

        if(CPCEXPLORER_STAGE4F2_VARIANT STREQUAL "OPT_DISPLAY" AND
           (NOT CPCEXPLORER_FAST_RAM_ACCESS OR NOT CPCEXPLORER_FAST_Z80_SRAM OR CPCEXPLORER_STAGE4F1_EXPERIMENT))
